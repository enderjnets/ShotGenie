/// One horizontal strip of the genie funnel.
public static func strip(v0: CGFloat, v1: CGFloat, rect: CGRect,
                         target: CGPoint, curve: Double, slide: Double) -> Strip? {
    let travel = target.y - rect.minY
    let offset = slide * travel
    let y0 = rect.minY + v0 * rect.height + offset
    let y1 = rect.minY + v1 * rect.height + offset
    guard y0 < target.y else { return nil }

    // Narrow toward the icon with a smoothstep.
    func edges(_ y: CGFloat) -> (CGFloat, CGFloat) {
        let u = smooth((y - rect.minY) / (target.y - rect.minY)) * curve
        let left  = rect.minX + (target.x - Genie.targetWidth / 2 - rect.minX) * u
        let right = rect.maxX + (target.x + Genie.targetWidth / 2 - rect.maxX) * u
        return (left, right)
    }
    let (l0, r0) = edges(y0), (l1, r1) = edges(min(y1, target.y))
    return Strip(dest: CGRect(x: min(l0, l1), y: y0, width: max(r0, r1) - min(l0, l1), height: y1 - y0),
                 image: imageRect(v0, v1, rect))
}
